function escapeHtml(str) {
return str.replace(/[&<>"']/g, m => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]));
        }
function highlightText(text, query) {
if (!query) return escapeHtml(text);
const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
const regex = new RegExp(`(${escaped})`, 'gi');
return escapeHtml(text).replace(regex, '<mark>$1</mark>');
        }

function setModifierKeyBlinking(filter) {
const keysToBlink = [];
if (filter === '⌘') { keysToBlink.push('command'); }
else if (filter === '⌘⇧') { keysToBlink.push('command', 'shift'); }
else if (filter === '⌘⌥') { keysToBlink.push('command', 'option'); }
else if (filter === '⌘⇧⌥') { keysToBlink.push('command', 'shift', 'option'); }
else if (filter === '⇧') { keysToBlink.push('shift'); }
else if (filter === '⌃') { keysToBlink.push('control'); }
keys.forEach(key => {
const keyCode = key.getAttribute('data-key');
const shouldBlink = keysToBlink.includes(keyCode);
if (key.classList.contains('modifier')) {
if (shouldBlink) {
if (key.classList.contains('blinking')) {
key.style.animation = 'none';
void key.offsetHeight;
key.style.animation = '';
                        } else { key.classList.add('blinking'); }
                    } else { key.classList.remove('blinking'); }
                }
            });
        }
function markKeysWithShortcuts() {
keys.forEach(key => {
const keyCode = key.getAttribute('data-key');
if (!keyCode) return;
const hasShortcut = Object.keys(shortcuts).some(shortcut => {
const shortcutKey = shortcut.split(/[⌘⇧⌥⌃]/).pop();
return keyMapping[shortcutKey] === keyCode;
                });
if (hasShortcut) key.classList.add('has-shortcut');
            });
        }
function getKeysForShortcut(shortcut) {
const keysToHighlight = [];
if (shortcut.includes('⌘')) keysToHighlight.push('command');
if (shortcut.includes('⇧')) keysToHighlight.push('shift');
if (shortcut.includes('⌥')) keysToHighlight.push('option');
if (shortcut.includes('⌃')) keysToHighlight.push('control');
const mainKey = shortcut.split(/[⌘⇧⌥⌃]/).pop();
if (keyMapping[mainKey]) keysToHighlight.push(keyMapping[mainKey]);
else if (mainKey) keysToHighlight.push(mainKey);
return keysToHighlight;
        }
function getAllKeysForFilter(filter) {
const allKeys = new Set();
for (let [shortcut] of Object.entries(shortcuts)) {
let matches = false;
if (filter === '⌘') matches = shortcut.startsWith('⌘') && !shortcut.includes('⇧') && !shortcut.includes('⌥');
else if (filter === '⌘⇧') matches = shortcut.includes('⌘⇧') && !shortcut.includes('⌥');
else if (filter === '⌘⌥') matches = shortcut.includes('⌘⌥') && !shortcut.includes('⇧');
else if (filter === '⌘⇧⌥') matches = shortcut.includes('⌘⇧⌥');
else if (filter === '⇧') matches = shortcut.startsWith('⇧') && !shortcut.includes('⌘');
else if (filter === '⌃') matches = shortcut.startsWith('⌃');
else if (filter === '単体') matches = !shortcut.includes('⌘') && !shortcut.includes('⇧') && !shortcut.includes('⌥') && !shortcut.includes('⌃');
if (matches) getKeysForShortcut(shortcut).forEach(k => allKeys.add(k));
            }
return Array.from(allKeys);
        }

function getShortcutsForKey(keyCode, filter = null) {
const result = [];
for (let [shortcut, description] of Object.entries(shortcuts)) {
if (filter && filter !== 'ALL') {
let matchesFilter = false;
if (filter === '⌘') matchesFilter = shortcut.startsWith('⌘') && !shortcut.includes('⇧') && !shortcut.includes('⌥');
else if (filter === '⌘⇧') matchesFilter = shortcut.includes('⌘⇧') && !shortcut.includes('⌥');
else if (filter === '⌘⌥') matchesFilter = shortcut.includes('⌘⌥') && !shortcut.includes('⇧');
else if (filter === '⌘⇧⌥') matchesFilter = shortcut.includes('⌘⇧⌥');
else if (filter === '⇧') matchesFilter = shortcut.startsWith('⇧') && !shortcut.includes('⌘');
else if (filter === '⌃') matchesFilter = shortcut.startsWith('⌃');
else if (filter === '単体') matchesFilter = !shortcut.includes('⌘') && !shortcut.includes('⇧') && !shortcut.includes('⌥') && !shortcut.includes('⌃');
if (!matchesFilter) continue;

const mainKey = shortcut.split(/[⌘⇧⌥⌃]/).pop();

function showTooltip(keyElement, shortcutsForKey) {
if (shortcutsForKey.length === 0) return;
if (tooltipTimeout) clearTimeout(tooltipTimeout);
tooltipShortcut.textContent = shortcutsForKey.map(s => s.shortcut).join(', ');
tooltipDescription.textContent = shortcutsForKey.map(s => s.description).join(' / ');
const keyRect = keyElement.getBoundingClientRect();
const tooltipRect = tooltip.getBoundingClientRect();
let left = keyRect.left + (keyRect.width / 2) - (tooltipRect.width / 2);
let top = keyRect.bottom + 15;
if (left < 10) left = 10;
if (left + tooltipRect.width > window.innerWidth - 10) left = window.innerWidth - tooltipRect.width - 10;
tooltip.style.left = left + 'px';
tooltip.style.top = top + 'px';
tooltip.classList.add('show');
        }
function hideTooltip() {
tooltip.classList.remove('show');
        }
function displayShortcutList(filter, buttonText) {
const filteredShortcuts = getShortcutsForFilter(filter);
const html = `
                <div class="shortcut-list">
                    <div class="list-header">
                        <div class="list-title">${buttonText}</div>
                        <div class="list-count">${filteredShortcuts.length}個</div>
                    </div>
                    <div class="shortcut-items">
${filteredShortcuts.map(item => `
                            <div class="shortcut-item" data-shortcut="${item.shortcut}">
                                <span class="shortcut-key">${item.shortcut}</span>
                                <span class="shortcut-desc">${item.description}</span>
                            </div>
                        `).join('')}
                    </div>
                </div>
            `;
listContainer.innerHTML = html;
const filterItemsEl = listContainer.querySelector('.shortcut-items');
if (filterItemsEl) {
filterItemsEl.addEventListener('mouseover', (e) => {
const item = e.target.closest('.shortcut-item');
if (!item) return;
filterItemsEl.querySelectorAll('.shortcut-item').forEach(i => i.classList.remove('highlighted'));
item.classList.add('highlighted');
highlightKeys(getKeysForShortcut(item.getAttribute('data-shortcut')));
                });
filterItemsEl.addEventListener('mouseout', (e) => {
if (filterItemsEl.contains(e.relatedTarget)) return;
filterItemsEl.querySelectorAll('.shortcut-item').forEach(i => i.classList.remove('highlighted'));
                });
            }
const shortcutList = document.querySelector('.shortcut-list');
if (shortcutList) {
shortcutList.addEventListener('mouseenter', () => { isInListArea = true; clearHighlights(); });
shortcutList.addEventListener('mouseleave', () => { isInListArea = false; restoreCategoryHighlight(); });
            }
        }
function displayShortcutsForSingleKey(keyCode, keyLabel) {
const allShortcuts = getShortcutsForKey(keyCode, null);
const html = `
                <div class="shortcut-list">
                    <div class="list-header">
                        <div class="list-title">${keyLabel} キーのショートカット</div>
                        <div class="list-count">${allShortcuts.length}個</div>
                    </div>
                    <div class="shortcut-items" id="singleKeyItems">
${allShortcuts.map(item => `
                            <div class="shortcut-item" data-shortcut="${item.shortcut}">
                                <span class="shortcut-key">${item.shortcut}</span>
                                <span class="shortcut-desc">${item.description}</span>
                            </div>
                        `).join('')}
                    </div>
                </div>
            `;
listContainer.innerHTML = html;
const singleKeyItemsEl = document.getElementById('singleKeyItems');
if (singleKeyItemsEl) {
singleKeyItemsEl.addEventListener('mouseover', (e) => {
const item = e.target.closest('.shortcut-item');
if (!item) return;
singleKeyItemsEl.querySelectorAll('.shortcut-item').forEach(i => i.classList.remove('highlighted'));
item.classList.add('highlighted');
highlightKeys(getKeysForShortcut(item.getAttribute('data-shortcut')));
                });
singleKeyItemsEl.addEventListener('mouseout', (e) => {
if (singleKeyItemsEl.contains(e.relatedTarget)) return;
singleKeyItemsEl.querySelectorAll('.shortcut-item').forEach(i => i.classList.remove('highlighted'));
                });
            }
const shortcutList2 = document.querySelector('.shortcut-list');
if (shortcutList2) {
shortcutList2.addEventListener('mouseenter', () => { isInListArea = true; clearHighlights(); });
shortcutList2.addEventListener('mouseleave', () => {
isInListArea = false;
if (isALLMode && !currentSearchQuery) highlightKeys([keyCode]);
                });
            }
        }
// キーボードエリア再入
keyboard.addEventListener('mouseenter', () => {
if (currentFilter && currentFilter !== 'ALL' && !isInListArea) restoreCategoryHighlight();
        });
// フィルターボタン
filterButtons.forEach(btn => {
btn.addEventListener('click', () => {
const filter = btn.getAttribute('data-filter');
if (currentActiveButton === btn) {
btn.classList.remove('active');
currentActiveButton = null;
isHighlightActive = false;
currentFilter = null;
isALLMode = false;
listContainer.innerHTML = '';
clearHighlights();
setModifierKeyBlinking(null);
// 検索エリアを隠す
searchArea.classList.remove('visible');
searchInput.value = '';
searchClear.classList.remove('show');
currentSearchQuery = '';
                } else {
filterButtons.forEach(b => b.classList.remove('active'));
btn.classList.add('active');
currentActiveButton = btn;
isHighlightActive = true;
currentFilter = filter;
if (filter === 'ALL') {
isALLMode = true;
clearHighlights();
listContainer.innerHTML = '';
setModifierKeyBlinking(null);
// 検索エリアを表示
searchArea.classList.add('visible');
// 既存のクエリがあれば再検索
if (currentSearchQuery) displaySearchResults(currentSearchQuery);
                    } else {
isALLMode = false;
searchArea.classList.remove('visible');
searchInput.value = '';
searchClear.classList.remove('show');
currentSearchQuery = '';
if (!isInListArea) highlightKeys(getAllKeysForFilter(filter));
setModifierKeyBlinking(filter);
displayShortcutList(filter, btn.textContent.trim());
                    }
                }
            });
        });
// キーホバー（検索中は無視）
keys.forEach(key => {
key.addEventListener('mouseenter', () => {
const keyCode = key.getAttribute('data-key');
if (!keyCode) return;
// 検索中はキーホバーによる上書きをしない
if (currentSearchQuery) return;
if (isALLMode) {
highlightKeys([keyCode]);
displayShortcutsForSingleKey(keyCode, key.textContent.trim());
                } else if (isHighlightActive && key.classList.contains('active-shortcut') && !key.classList.contains('modifier')) {
const shortcutsForKey = getShortcutsForKey(keyCode, currentFilter);
if (shortcutsForKey.length > 0) showTooltip(key, shortcutsForKey);
                }
            });
key.addEventListener('mouseleave', () => {
hideTooltip();
if (currentSearchQuery) return; // 検索中は状態を保持
if (isALLMode) {
clearHighlights();
listContainer.innerHTML = '';
                } else if (currentFilter && currentFilter !== 'ALL' && !isInListArea) {
restoreCategoryHighlight();
                }
            });
        });
document.addEventListener('click', (e) => {
if (!e.target.closest('.key') && !e.target.closest('.key-tooltip')) hideTooltip();
        });
markKeysWithShortcuts();
window.addEventListener('DOMContentLoaded', () => {
const allButton = document.querySelector('.filter-btn[data-filter="ALL"]');
if (allButton) allButton.click();
        });
</script>
