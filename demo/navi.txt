function chipRow(box, items, onPick){
  items.forEach(it=>{
    const b=document.createElement('button');
    b.className='chip';
    b.innerHTML=`${it.name}<small>${it.hint||''}</small>`;
    b.onclick=()=>{[...box.children].forEach(c=>c.classList.toggle('on',c===b)); onPick(it);};
    box.appendChild(b);
  });
}

}
