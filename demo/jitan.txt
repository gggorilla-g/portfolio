<script>
const CAT_COLORS = {
  OCR: '#1D9E75', GAS: '#185FA5', Python: '#BA7517',
  HTML: '#534AB7', ExtendScript: '#D85A30', 'その他': '#888780'
};

const DEFAULTS = [
  {id:1, name:'MATE（名刺OCR）',   cat:'OCR',    before:20, after:2, freq:60},
  {id:2, name:'BASE（BGM自動化）', cat:'GAS',    before:45, after:5, freq:30},
  {id:3, name:'Time is Money',     cat:'OCR',    before:90, after:8, freq:8},
  {id:4, name:'BannerSpec',        cat:'HTML',   before:30, after:5, freq:40},
  {id:5, name:'Excel→AI自動組版', cat:'Python', before:60, after:5, freq:20},
];
const LS_KEY = 'jitan_tracker_v1';

function loadState() {
  try {
    const s = JSON.parse(localStorage.getItem(LS_KEY));
    if (s && Array.isArray(s.tools)) return s;
  } catch(e) {}
  return null;
}

function persist() {
  try {
    localStorage.setItem(LS_KEY, JSON.stringify({ tools, nid, view }));
  } catch(e) {}
}

const saved = loadState();
let tools = saved ? saved.tools : DEFAULTS;
let nid   = saved ? saved.nid  : 6;
let view  = saved ? saved.view : 'month';

function mult() { return view === 'year' ? 12 : 1; }
function saveMins(t) { return Math.max(0, (t.before - t.after)) * t.freq; }
function toH(mins) { return Math.round(mins / 60 * 10) / 10; }
function eff(t) { return t.before > 0 ? Math.round((t.before - t.after) / t.before * 100) : 0; }

function animateTo(el, target, decimals) {
  const start = parseFloat(el.textContent.replace(/,/g, '')) || 0;
  const steps = 24; let i = 0;
  const step = () => {
    i++;
    const v = start + (target - start) * (i / steps);
    el.textContent = decimals === 0 ? Math.round(v).toLocaleString() : v.toFixed(decimals);
    if (i < steps) requestAnimationFrame(step);
    else el.textContent = decimals === 0 ? Math.round(target).toLocaleString() : target.toFixed(decimals);
  };
  requestAnimationFrame(step);
}

function render() {
  const m = mult();
  const totalMins = tools.reduce((s, t) => s + saveMins(t), 0) * m;
  const totalH = toH(totalMins);
  const days = Math.round(totalH / 8 * 100) / 100;

  animateTo(document.getElementById('hero-num'), totalH, 1);
  document.getElementById('hero-label').textContent = (view === 'year' ? '年間' : '月間') + ' 削減時間';
  document.getElementById('hero-sub').textContent = tools.length
    ? `${tools.length}本のツールで ${Math.round(totalMins).toLocaleString()} 分を削減`
    : 'ツールを追加してください';

  animateTo(document.getElementById('days-val'), days, 2);
  document.getElementById('days-lbl').textContent = view === 'year' ? '年換算稼働日（8h）' : '月換算稼働日（8h）';
  animateTo(document.getElementById('min-val'), Math.round(totalMins), 0);
  document.getElementById('min-lbl').textContent = view === 'year' ? '年間削減 分' : '月間削減 分';
  document.getElementById('tools-val').textContent = tools.length;

  document.getElementById('btn-m').classList.toggle('on', view === 'month');
  document.getElementById('btn-y').classList.toggle('on', view === 'year');

  const list = document.getElementById('tool-list');
  if (!tools.length) {
    list.innerHTML = '<div class="empty"><i class="ti ti-robot" style="font-size:24px;display:block;margin-bottom:8px" aria-hidden="true"></i>ツールを追加してください</div>';
    document.getElementById('bar-section').style.display = 'none';
    return;
  }

  list.innerHTML = tools.map(t => {
    const sm = saveMins(t);
    const dispH = toH(sm * m);
    const dispMins = Math.round(sm * m);
    const e = eff(t);
    const color = CAT_COLORS[t.cat] || '#888780';
    const chipColor = dispH > 0 ? color : 'var(--text3)';
    return `<div class="tool-card">
      <div class="tool-card-head">
        <div class="tool-name-wrap">
          <div class="tool-name">${t.name}</div>
          <div class="tool-cat">${t.cat}</div>
        </div>
        <div style="text-align:right;flex-shrink:0">
          <div class="result-chip" style="color:${chipColor}">${dispH > 0 ? '+' + dispH + 'h' : '–'}</div>
          <div class="result-sub">${dispMins > 0 ? dispMins.toLocaleString() + '分' : ''} ${e > 0 ? '削減率 ' + e + '%' : ''}</div>
        </div>
        <button class="del-btn" onclick="del(${t.id})" aria-label="削除"><i class="ti ti-trash" style="font-size:14px" aria-hidden="true"></i></button>
      </div>
      <div class="tool-card-inputs">
        <div class="inp-cell">
          <div class="inp-lbl">導入前（分/回）</div>
          <div class="inp-wrap">
            <input class="n-inp" type="number" min="0" step="1" value="${t.before}" onchange="updField(${t.id},'before',this.value)">
            <span class="inp-unit">分</span>
          </div>
        </div>
        <div class="inp-cell">
          <div class="inp-lbl">導入後（分/回）</div>
          <div class="inp-wrap">
            <input class="n-inp" type="number" min="0" step="1" value="${t.after}" onchange="updField(${t.id},'after',this.value)">
            <span class="inp-unit">分</span>
          </div>
        </div>
        <div class="inp-cell">
          <div class="inp-lbl">月間稼働数</div>
          <div class="inp-wrap">
            <input class="n-inp" type="number" min="0" step="1" value="${t.freq}" onchange="updField(${t.id},'freq',this.value)">
            <span class="inp-unit">回</span>
          </div>
        </div>
      </div>
    </div>`;
  }).join('');

  const barSec = document.getElementById('bar-section');
  barSec.style.display = 'block';
  const sorted = [...tools].sort((a, b) => saveMins(b) - saveMins(a));
  const maxV = saveMins(sorted[0]) * m || 1;
  document.getElementById('bars').innerHTML = sorted.map(t => {
    const v = toH(saveMins(t) * m);
    const pct = Math.round(saveMins(t) * m / maxV * 100);
    const color = CAT_COLORS[t.cat] || '#888780';
    return `<div class="bar-row">
      <span class="bar-name">${t.name}</span>
      <div class="bar-track"><div class="bar-fill" style="width:${pct}%;background:${color}"></div></div>
      <span class="bar-val">${v}h</span>
    </div>`;
  }).join('');
}

function setView(v) { view = v; persist(); render(); }
function updField(id, field, val) {
  const t = tools.find(t => t.id === id);
  if (t) { t[field] = Math.max(0, parseInt(val) || 0); persist(); render(); }
}
function del(id) { tools = tools.filter(t => t.id !== id); persist(); render(); }

function openModal() { document.getElementById('modal').classList.add('open'); }
function closeModal() {
  document.getElementById('modal').classList.remove('open');
  ['f-name','f-before','f-after','f-freq'].forEach(id => document.getElementById(id).value = '');
}
function saveModal() {
  const name = document.getElementById('f-name').value.trim();
  if (!name) return;
  tools.push({
    id: nid++,
    name,
    cat: document.getElementById('f-cat').value,
    before: Math.max(0, parseInt(document.getElementById('f-before').value) || 0),
    after:  Math.max(0, parseInt(document.getElementById('f-after').value)  || 0),
    freq:   Math.max(0, parseInt(document.getElementById('f-freq').value)   || 0),
  });
  persist();
  closeModal();
  render();
}

document.getElementById('modal').addEventListener('click', e => {
  if (e.target === document.getElementById('modal')) closeModal();
});

render();
</script>
</body>
